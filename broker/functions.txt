async function summaryFetchCanonicalRow(section, id, ctx) {
  section = String(section || '').trim();
  id      = String(id || '').trim();
  if (!section || !id) return null;

  const ctxObj = (ctx && typeof ctx === 'object') ? ctx : {};
  const forceFresh = !!(
    ctxObj.forceFresh ||
    ctxObj.bypassCache ||
    ctxObj.skipCache ||
    ctxObj.invalidateCache ||
    ctxObj.refreshCanonical
  );

  const wantsTimesheetLifecycleFallback = section === 'timesheets' && !!(
    ctxObj.lifecycleFallback === true ||
    ctxObj.lifecycle_fallback === true ||
    ctxObj.unfilteredById === true ||
    ctxObj.unfiltered_by_id === true ||
    ctxObj.afterLifecycleMutation === true ||
    ctxObj.after_lifecycle_mutation === true
  );

  const sameCanonicalTimesheetId = (row) => {
    if (!row || typeof row !== 'object' || Array.isArray(row)) return false;
    const rowKeyCandidates = [
      row.row_key,
      row.rowKey,
      row.new_row_key,
      row.newRowKey,
      row.identity && row.identity.row_key,
      row.identity && row.identity.rowKey,
      row.identity && row.identity.current_row_key,
      row.identity && row.identity.currentRowKey
    ].map((value) => String(value || '').trim()).filter(Boolean);
    if (rowKeyCandidates.includes(`timesheet:${id}`) || rowKeyCandidates.includes(`contract_week:${id}`) || rowKeyCandidates.includes(`contract-week:${id}`)) return true;

    const candidates = [
      row.id,
      row.timesheet_id,
      row.current_timesheet_id,
      row.requested_timesheet_id,
      row.expected_timesheet_id,
      row.previous_timesheet_id,
      row.contract_week_id,
      row.identity && row.identity.timesheet_id,
      row.identity && row.identity.current_timesheet_id,
      row.identity && row.identity.requested_timesheet_id,
      row.identity && row.identity.expected_timesheet_id,
      row.identity && row.identity.previous_timesheet_id,
      row.identity && row.identity.contract_week_id
    ].map((value) => String(value || '').trim()).filter(Boolean);
    return candidates.includes(id);
  };

  const normaliseCanonicalTimesheetRow = (row, source) => {
    if (!row || typeof row !== 'object' || Array.isArray(row)) return null;

    const isPlainObject = (value) => !!(value && typeof value === 'object' && !Array.isArray(value));
    const trimLocal = (value) => String(value == null ? '' : value).trim();
    const firstNonBlank = (...values) => {
      for (const value of values) {
        const text = trimLocal(value);
        if (text) return text;
      }
      return '';
    };
    const firstDefined = (...values) => {
      for (const value of values) {
        if (value !== undefined && value !== null && trimLocal(value) !== '') return value;
      }
      return undefined;
    };
    const boolishNullable = (value) => {
      if (value === true || value === 1) return true;
      if (value === false || value === 0) return false;
      if (value == null || value === '') return null;
      const raw = trimLocal(value).toLowerCase();
      if (!raw) return null;
      if (['true', 't', '1', 'yes', 'y', 'on', 'authorised', 'authorized'].includes(raw)) return true;
      if (['false', 'f', '0', 'no', 'n', 'off', 'unauthorised', 'unauthorized', 'pending_auth'].includes(raw)) return false;
      return null;
    };
    const assignIfPresent = (target, targetKey, ...values) => {
      const value = firstDefined(...values);
      if (value !== undefined) target[targetKey] = value;
    };

    const out = Object.assign({}, row || {});
    const identity = isPlainObject(out.identity) ? out.identity : {};
    const status = isPlainObject(out.status) ? out.status : {};
    const actions = isPlainObject(out.actions) ? out.actions : {};
    const display = isPlainObject(out.display) ? out.display : {};
    const actionFlags = isPlainObject(out.action_flags) ? out.action_flags : (isPlainObject(out.actionFlags) ? out.actionFlags : {});
    const editHints = isPlainObject(out.edit_permission_hints) ? out.edit_permission_hints : (isPlainObject(out.editPermissionHints) ? out.editPermissionHints : {});
    const footerHints = isPlainObject(out.footer_action_hints) ? out.footer_action_hints : (isPlainObject(out.footerActionHints) ? out.footerActionHints : {});
    const priorityHints = isPlainObject(out.priority_badge_hints) ? out.priority_badge_hints : (isPlainObject(out.priorityBadgeHints) ? out.priorityBadgeHints : {});
    const evidenceSummary = isPlainObject(out.evidence_summary) ? out.evidence_summary : {};
    const stagedEvidenceSummary = isPlainObject(out.staged_evidence_summary) ? out.staged_evidence_summary : {};

    const currentTimesheetId = firstNonBlank(
      out.current_timesheet_id,
      out.currentTimesheetId,
      out.timesheet_id,
      out.timesheetId,
      identity.current_timesheet_id,
      identity.currentTimesheetId,
      identity.timesheet_id,
      identity.timesheetId
    );
    const requestedTimesheetId = firstNonBlank(
      out.requested_timesheet_id,
      out.requestedTimesheetId,
      identity.requested_timesheet_id,
      identity.requestedTimesheetId,
      currentTimesheetId
    );
    const expectedTimesheetId = firstNonBlank(
      out.expected_timesheet_id,
      out.expectedTimesheetId,
      identity.expected_timesheet_id,
      identity.expectedTimesheetId,
      currentTimesheetId
    );
    const previousTimesheetId = firstNonBlank(
      out.previous_timesheet_id,
      out.previousTimesheetId,
      identity.previous_timesheet_id,
      identity.previousTimesheetId
    );
    const contractWeekId = firstNonBlank(
      out.contract_week_id,
      out.contractWeekId,
      identity.contract_week_id,
      identity.contractWeekId,
      identity.week_id,
      identity.weekId
    );
    const rowKey = firstNonBlank(
      out.row_key,
      out.rowKey,
      out.new_row_key,
      out.newRowKey,
      identity.row_key,
      identity.rowKey,
      currentTimesheetId ? `timesheet:${currentTimesheetId}` : '',
      contractWeekId ? `contract_week:${contractWeekId}` : ''
    );
    const previousRowKey = firstNonBlank(
      out.previous_row_key,
      out.previousRowKey,
      out.old_row_key,
      out.oldRowKey,
      identity.previous_row_key,
      identity.previousRowKey,
      previousTimesheetId ? `timesheet:${previousTimesheetId}` : ''
    );
    const bookingId = firstNonBlank(out.booking_id, out.bookingId, identity.booking_id, identity.bookingId, display.booking_id, display.bookingId);

    if (currentTimesheetId) {
      out.timesheet_id = currentTimesheetId;
      out.current_timesheet_id = currentTimesheetId;
    }
    if (requestedTimesheetId) out.requested_timesheet_id = requestedTimesheetId;
    if (expectedTimesheetId) out.expected_timesheet_id = expectedTimesheetId;
    if (previousTimesheetId) out.previous_timesheet_id = previousTimesheetId;
    if (contractWeekId) out.contract_week_id = contractWeekId;
    if (bookingId) out.booking_id = bookingId;
    if (rowKey) out.row_key = rowKey;
    if (previousRowKey) out.previous_row_key = previousRowKey;
    if (!out.id) out.id = currentTimesheetId || contractWeekId || id;

    const backendSignature = firstNonBlank(
      out.backend_row_signature,
      out.backendRowSignature,
      out.expected_row_signature,
      out.expectedRowSignature,
      out.row_signature,
      out.rowSignature,
      out.mutation_row_signature,
      out.mutationRowSignature,
      out.render_signature,
      out.renderSignature,
      identity.backend_row_signature,
      identity.backendRowSignature,
      identity.row_signature,
      identity.rowSignature
    );
    if (backendSignature) {
      out.backend_row_signature = backendSignature;
      out.row_signature = backendSignature;
      out.expected_row_signature = firstNonBlank(out.expected_row_signature, out.expectedRowSignature, backendSignature);
      out.mutation_row_signature = firstNonBlank(out.mutation_row_signature, out.mutationRowSignature, backendSignature);
      out.render_signature = firstNonBlank(out.render_signature, out.renderSignature, backendSignature);
    }

    const processingStatus = firstNonBlank(
      out.processing_status,
      out.processingStatus,
      out.new_processing_status,
      out.newProcessingStatus,
      status.processing_status,
      status.processingStatus,
      priorityHints.processing_status,
      priorityHints.processingStatus
    );
    const stateAfter = firstNonBlank(out.state_after, out.stateAfter, status.state_after, status.stateAfter, priorityHints.state_after, priorityHints.stateAfter);
    const statusBucket = firstNonBlank(status.bucket, status.bulk_process_bucket, out.bulk_process_bucket, out.bulkProcessBucket);
    const summaryStage = firstNonBlank(
      out.summary_stage,
      out.summaryStage,
      out.tools_stage,
      out.toolsStage,
      status.summary_stage,
      status.summaryStage,
      statusBucket,
      display.summary_stage,
      display.summaryStage,
      priorityHints.summary_stage,
      priorityHints.summaryStage
    );
    if (processingStatus) out.processing_status = processingStatus;
    if (stateAfter) out.state_after = stateAfter;
    if (summaryStage) {
      out.summary_stage = summaryStage;
      out.tools_stage = firstNonBlank(out.tools_stage, out.toolsStage, summaryStage);
    }
    if (statusBucket) out.bulk_process_bucket = statusBucket;
    assignIfPresent(out, 'contract_week_status', out.contract_week_status, out.contractWeekStatus, status.contract_week_status, status.contractWeekStatus);

    let authorisedValue = boolishNullable(firstDefined(
      out.is_authorised,
      out.isAuthorised,
      out.authorised,
      out.authorized,
      status.is_authorised,
      status.isAuthorised,
      status.authorised,
      status.authorized,
      actionFlags.is_authorised,
      actionFlags.isAuthorised,
      actionFlags.authorised,
      footerHints.is_authorised,
      footerHints.authorised,
      priorityHints.is_authorised,
      priorityHints.isAuthorised,
      priorityHints.authorised
    ));
    const actionUpper = firstNonBlank(out.action, out.lifecycle_action, out.lifecycleAction).toUpperCase();
    const stateUpper = stateAfter.toUpperCase();
    const processingUpper = processingStatus.toUpperCase();
    if (authorisedValue === null) {
      if (stateUpper === 'AUTHORISED' || stateUpper === 'AUTHORIZED' || actionUpper === 'AUTHORISE' || actionUpper === 'AUTHORIZE') authorisedValue = true;
      else if (stateUpper === 'UNAUTHORISED' || stateUpper === 'UNAUTHORIZED' || actionUpper === 'UNAUTHORISE' || actionUpper === 'UNAUTHORIZE' || processingUpper === 'PENDING_AUTH') authorisedValue = false;
      else if (firstNonBlank(out.authorised_at_server, out.authorisedAtServer, out.authorised_at_utc, out.authorisedAtUtc, status.authorised_at_server, status.authorisedAtServer, status.authorised_at_utc, status.authorisedAtUtc)) authorisedValue = true;
    }
    if (authorisedValue !== null) {
      out.is_authorised = authorisedValue;
      out.authorised = authorisedValue;
      out.review_only = boolishNullable(firstDefined(out.review_only, out.reviewOnly, editHints.read_only, editHints.readOnly, actionFlags.review_only, actionFlags.reviewOnly)) ?? authorisedValue;
    }
    assignIfPresent(out, 'authorised_at_server', out.authorised_at_server, out.authorisedAtServer, status.authorised_at_server, status.authorisedAtServer);
    assignIfPresent(out, 'authorised_at_utc', out.authorised_at_utc, out.authorisedAtUtc, status.authorised_at_utc, status.authorisedAtUtc);
    if (authorisedValue === false) {
      out.authorised_at_server = null;
      out.authorised_at_utc = null;
    }

    const readyToPayValue = boolishNullable(firstDefined(out.ready_to_pay, out.readyToPay, status.ready_to_pay, status.readyToPay, display.ready_to_pay, display.readyToPay, priorityHints.ready_to_pay, priorityHints.readyToPay));
    if (readyToPayValue !== null) out.ready_to_pay = readyToPayValue;
    [
      ['is_paid', status.is_paid, status.isPaid, out.is_paid, out.isPaid, priorityHints.is_paid, priorityHints.isPaid],
      ['is_invoice_locked', status.is_invoice_locked, status.isInvoiceLocked, out.is_invoice_locked, out.isInvoiceLocked],
      ['is_locked', status.is_locked, status.isLocked, status.locked, out.is_locked, out.isLocked, out.locked],
      ['qr_unsigned_blocked', status.qr_unsigned_blocked, status.qrUnsignedBlocked, out.qr_unsigned_blocked, out.qrUnsignedBlocked]
    ].forEach(([key, ...values]) => {
      const parsed = boolishNullable(firstDefined(...values));
      if (parsed !== null) out[key] = parsed;
    });

    const firstBool = (...values) => boolishNullable(firstDefined(...values));
    const canProcess = firstBool(actions.can_process, actions.canProcess, actionFlags.can_process, actionFlags.canProcess, out.can_process, out.canProcess);
    const canUnprocess = firstBool(actions.can_unprocess, actions.canUnprocess, actionFlags.can_unprocess, actionFlags.canUnprocess, out.can_unprocess, out.canUnprocess);
    const canAuthorise = firstBool(actions.can_authorise, actions.canAuthorise, actions.can_authorize, actionFlags.can_authorise, actionFlags.canAuthorise, actionFlags.can_authorize, footerHints.can_authorise, footerHints.canAuthorise, priorityHints.can_authorise, priorityHints.canAuthorise, out.can_authorise, out.canAuthorise, out.can_authorize);
    const canUnauthorise = firstBool(actions.can_unauthorise, actions.canUnauthorise, actions.can_unauthorize, actionFlags.can_unauthorise, actionFlags.canUnauthorise, actionFlags.can_unauthorize, footerHints.can_unauthorise, footerHints.canUnauthorise, priorityHints.can_unauthorise, priorityHints.canUnauthorise, out.can_unauthorise, out.canUnauthorise, out.can_unauthorize);
    if (canProcess !== null) out.can_process = canProcess;
    if (canUnprocess !== null) out.can_unprocess = canUnprocess;
    if (canAuthorise !== null) {
      out.can_authorise = canAuthorise;
      out.can_bulk_authorise = canAuthorise;
    }
    if (canUnauthorise !== null) {
      out.can_unauthorise = canUnauthorise;
      out.can_bulk_unauthorise = canUnauthorise;
    }

    const disabledReasons = [
      ...(Array.isArray(actions.disabled_reasons) ? actions.disabled_reasons : []),
      ...(Array.isArray(actions.disabledReasons) ? actions.disabledReasons : []),
      ...(Array.isArray(actionFlags.disabled_reasons) ? actionFlags.disabled_reasons : []),
      ...(Array.isArray(actionFlags.disabledReasons) ? actionFlags.disabledReasons : []),
      ...(Array.isArray(footerHints.disabled_reasons) ? footerHints.disabled_reasons : []),
      ...(Array.isArray(footerHints.disabledReasons) ? footerHints.disabledReasons : []),
      ...(Array.isArray(priorityHints.disabled_reasons) ? priorityHints.disabled_reasons : []),
      ...(Array.isArray(priorityHints.disabledReasons) ? priorityHints.disabledReasons : []),
      ...(Array.isArray(out.disabled_reasons) ? out.disabled_reasons : []),
      ...(Array.isArray(out.disabledReasons) ? out.disabledReasons : [])
    ].filter((value, index, arr) => trimLocal(value) && arr.findIndex((other) => trimLocal(other) === trimLocal(value)) === index);
    if (disabledReasons.length) out.disabled_reasons = disabledReasons;

    const mergedActionFlags = { ...actionFlags };
    if (Object.prototype.hasOwnProperty.call(out, 'can_process')) mergedActionFlags.can_process = out.can_process === true;
    if (Object.prototype.hasOwnProperty.call(out, 'can_unprocess')) mergedActionFlags.can_unprocess = out.can_unprocess === true;
    if (Object.prototype.hasOwnProperty.call(out, 'can_authorise')) {
      mergedActionFlags.can_authorise = out.can_authorise === true;
      mergedActionFlags.can_bulk_authorise = out.can_authorise === true;
    }
    if (Object.prototype.hasOwnProperty.call(out, 'can_unauthorise')) {
      mergedActionFlags.can_unauthorise = out.can_unauthorise === true;
      mergedActionFlags.can_bulk_unauthorise = out.can_unauthorise === true;
    }
    if (Object.prototype.hasOwnProperty.call(out, 'review_only')) mergedActionFlags.review_only = out.review_only === true;
    if (Array.isArray(out.disabled_reasons)) mergedActionFlags.disabled_reasons = out.disabled_reasons;
    if (Object.keys(mergedActionFlags).length) out.action_flags = mergedActionFlags;

    assignIfPresent(out, 'week_ending_date', out.week_ending_date, out.weekEndingDate, display.week_ending_date, display.weekEndingDate);
    assignIfPresent(out, 'candidate_id', out.candidate_id, out.candidateId, display.candidate_id, display.candidateId);
    assignIfPresent(out, 'candidate_name', out.candidate_name, out.candidateName, out.candidate_display_name, out.candidateDisplayName, display.candidate_display, display.candidateDisplay, display.candidate_display_name, display.candidateDisplayName);
    assignIfPresent(out, 'candidate_display_name', out.candidate_display_name, out.candidateDisplayName, out.candidate_name, display.candidate_display, display.candidateDisplay, display.candidate_display_name, display.candidateDisplayName);
    assignIfPresent(out, 'client_id', out.client_id, out.clientId, display.client_id, display.clientId);
    assignIfPresent(out, 'client_name', out.client_name, out.clientName, out.client_display_name, display.client_name, display.clientName);
    assignIfPresent(out, 'client_display_name', out.client_display_name, out.clientDisplayName, out.client_name, display.client_name, display.clientName);
    assignIfPresent(out, 'role', out.role, display.role);
    assignIfPresent(out, 'band', out.band, display.band);
    assignIfPresent(out, 'total_hours', out.total_hours, display.total_hours);
    assignIfPresent(out, 'total_pay_ex_vat', out.total_pay_ex_vat, display.total_pay_ex_vat);
    assignIfPresent(out, 'total_charge_ex_vat', out.total_charge_ex_vat, display.total_charge_ex_vat);
    assignIfPresent(out, 'margin_ex_vat', out.margin_ex_vat, display.margin_ex_vat);

    if (evidenceSummary.count != null || stagedEvidenceSummary.staged_count != null) {
      out.evidence_count = Number(evidenceSummary.count || 0) + Number(stagedEvidenceSummary.staged_count || 0);
      out.attached_evidence_count = Number(evidenceSummary.count || 0);
      out.queue_staged_count = Number(stagedEvidenceSummary.staged_count || 0);
    }
    if (firstNonBlank(evidenceSummary.primary_storage_key, evidenceSummary.manual_pdf_r2_key, stagedEvidenceSummary.primary_staged_timesheet_storage_key)) {
      out.primary_artifact_storage_key = firstNonBlank(evidenceSummary.primary_storage_key, evidenceSummary.manual_pdf_r2_key, stagedEvidenceSummary.primary_staged_timesheet_storage_key);
    }

    out.__canonical_fetch_source = source || out.__canonical_fetch_source || 'summaryFetchCanonicalRow';
    return out;
  };

  const fetchTimesheetLifecycleFallbackRow = async (reason) => {
    if (!wantsTimesheetLifecycleFallback) return null;

    const payload = {
      context: 'timesheet_modal',
      source: 'summaryFetchCanonicalRow-unfiltered-lifecycle-fallback',
      reason: reason || 'summary-fetch-empty-or-filtered',
      items: [{
        row_key: `timesheet:${id}`,
        timesheet_id: id,
        current_timesheet_id: id,
        requested_timesheet_id: id,
        expected_timesheet_id: id,
        context: 'timesheet_modal'
      }]
    };

    try {
      let json = null;
      if (typeof apiPostJson === 'function') {
        json = await apiPostJson('/api/timesheets/lifecycle-affected-rows', payload);
      } else if (typeof authFetch === 'function') {
        const fallbackUrl = (typeof API === 'function')
          ? API('/api/timesheets/lifecycle-affected-rows')
          : '/api/timesheets/lifecycle-affected-rows';
        const res = await authFetch(fallbackUrl, {
          method: 'POST',
          cache: 'no-store',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
        if (!res || !res.ok) return null;
        json = await res.json().catch(() => null);
      } else {
        return null;
      }

      const pools = [];
      if (Array.isArray(json?.rows)) json.rows.forEach((row) => pools.push(row));
      if (Array.isArray(json?.affected_rows)) json.affected_rows.forEach((row) => pools.push(row));
      if (Array.isArray(json?.items)) json.items.forEach((row) => pools.push(row));
      if (Array.isArray(json?.data)) json.data.forEach((row) => pools.push(row));
      if (Array.isArray(json?.results)) {
        json.results.forEach((res) => {
          if (Array.isArray(res?.rows)) res.rows.forEach((row) => pools.push(row));
          if (Array.isArray(res?.affected_rows)) res.affected_rows.forEach((row) => pools.push(row));
          if (Array.isArray(res?.items)) res.items.forEach((row) => pools.push(row));
          if (res?.row && typeof res.row === 'object' && !Array.isArray(res.row)) pools.push(res.row);
          if (res?.current_row && typeof res.current_row === 'object' && !Array.isArray(res.current_row)) pools.push(res.current_row);
        });
      }
      if (json?.row && typeof json.row === 'object' && !Array.isArray(json.row)) pools.push(json.row);
      if (json?.current_row && typeof json.current_row === 'object' && !Array.isArray(json.current_row)) pools.push(json.current_row);
      if (json?.lifecycle_patch && typeof json.lifecycle_patch === 'object' && !Array.isArray(json.lifecycle_patch)) {
        const patch = json.lifecycle_patch;
        if (patch.timesheet_id || patch.current_timesheet_id || patch.id || patch.contract_week_id) pools.push(patch);
      }

      const match = pools.find((row) => sameCanonicalTimesheetId(row));
      return match ? normaliseCanonicalTimesheetRow(match, 'lifecycle-affected-rows-unfiltered-fallback') : null;
    } catch (err) {
      try {
        console.warn('[TS][SUMMARY_CANONICAL][LIFECYCLE_FALLBACK_FAILED]', err?.message || err);
      } catch {}
      return null;
    }
  };

  // Related mode: do not attempt server canonical fetch unless this is an explicit
  // post-lifecycle timesheet fallback. For ordinary related lists, preserve the
  // original in-memory-only behaviour.
  const inferredRelated =
    !!(window.__listState &&
       window.__listState[section] &&
       window.__listState[section].filters &&
       window.__listState[section].filters.related);

  const inRelatedMode = !!(ctxObj.inRelatedMode || inferredRelated);

  if (inRelatedMode && !wantsTimesheetLifecycleFallback) {
    try {
      if (String(currentSection || '') !== section) return null;
      if (!Array.isArray(window.currentRows)) return null;

      const row = window.currentRows.find(r => {
        const rid = r && (r.id ?? r.timesheet_id ?? r.contract_week_id ?? r.invoice_id);
        return String(rid || '') === id;
      }) || null;

      return row && typeof row === 'object' ? Object.assign({}, row || {}) : null;
    } catch {
      return null;
    }
  }

  // Cache by section + fingerprint + id to avoid repeat calls. Lifecycle fallback
  // gets a distinct cache key so a filtered-summary null cannot mask a fresh row.
  const fp = (typeof getSummaryFingerprint === 'function')
    ? String(getSummaryFingerprint(section) || '')
    : (() => {
        const st0 = (window.__listState && window.__listState[section]) ? window.__listState[section] : {};
        const filters0 = st0 && st0.filters ? st0.filters : {};
        try { return JSON.stringify({ section, filters: filters0 }); } catch { return `${section}`; }
      })();

  window.__summaryCanonicalCache = window.__summaryCanonicalCache || {};
  const cacheKey = `${section}::${fp}::${id}${wantsTimesheetLifecycleFallback ? '::lifecycle' : ''}`;
  const now = Date.now();

  if (forceFresh) {
    try { delete window.__summaryCanonicalCache[cacheKey]; } catch {}
  }

  const hit = window.__summaryCanonicalCache[cacheKey];
  if (!forceFresh && hit && hit.t && (now - hit.t) < 8000) {
    return hit.row || null;
  }

  const st = (window.__listState && window.__listState[section]) ? window.__listState[section] : null;
  const filters = (st && st.filters && typeof st.filters === 'object') ? st.filters : {};
  const sort    = (st && st.sort && typeof st.sort === 'object') ? st.sort : { key: null, dir: 'asc' };

  // ── Selection-filter parity (avoid incorrect “included” fetch) ──────────────
  const parseIdSetFromFilters = (f) => {
    const out = new Set();

    const add = (v) => {
      const s = String(v || '').trim();
      if (s) out.add(s);
    };

    // ids can be array OR csv
    if (Object.prototype.hasOwnProperty.call(f, 'ids')) {
      const v = f.ids;
      if (Array.isArray(v)) v.forEach(add);
      else String(v || '').split(',').map(x => x.trim()).filter(Boolean).forEach(add);
    }

    // id can be a single uuid OR an in-list expression
    if (Object.prototype.hasOwnProperty.call(f, 'id')) {
      const v = String(f.id || '').trim();
      if (v) {
        if (/^in\.\(.+\)$/.test(v)) {
          const inner = v.replace(/^in\.\(/, '').replace(/\)$/, '');
          inner.split(',').map(x => x.trim()).filter(Boolean).forEach(add);
        } else {
          add(v);
        }
      }
    }

    return out;
  };

  try {
    const selIds = parseIdSetFromFilters(filters || {});
    const selectedIds = Array.isArray(st?.selectedIds)
      ? st.selectedIds.map((value) => String(value || '').trim()).filter(Boolean)
      : [];
    selectedIds.forEach((value) => selIds.add(value));

    if (selIds.size > 0 && !selIds.has(id)) {
      const fallback = await fetchTimesheetLifecycleFallbackRow('selected-id-filter-excluded-target-after-lifecycle');
      window.__summaryCanonicalCache[cacheKey] = { t: now, row: fallback || null };
      return fallback || null;
    }
  } catch {}

  const qs = new URLSearchParams();

  // Always request a single row
  qs.set('page', '1');
  qs.set('page_size', '1');

  const appendFilter = (k, v) => {
    if (v == null) return;
    if (typeof v === 'string' && v.trim() === '') return;

    if (Array.isArray(v)) {
      v.forEach(x => {
        if (x == null) return;
        const s = String(x).trim();
        if (!s) return;
        qs.append(k, s);
      });
      return;
    }

    qs.set(k, String(v));
  };

  const lifecycleStageFilterKeys = new Set([
    'tools_stage',
    'stage',
    'summary_stage',
    'bulk_authorise_stage',
    'bulk_authorize_stage',
    'bulk_process_stage',
    'bulk_process_bucket',
    'authorisation_stage',
    'authorization_stage',
    'processing_stage',
    'status_stage',
    'status',
    'processing_status',
    'authorisation_status',
    'authorization_status'
  ]);

  // Apply existing filters, but strip:
  // - related (handled elsewhere)
  // - id/ids (we already validated selection parity; we’ll set our own id fetch)
  // - pagination keys (we set page/page_size)
  // - stage/status filters only for explicit post-lifecycle timesheet fallback
  Object.entries(filters || {}).forEach(([k, v]) => {
    if (k === 'related') return;
    if (k === 'page' || k === 'page_size' || k === 'limit' || k === 'offset') return;
    if (k === 'id' || k === 'ids') return;
    if (wantsTimesheetLifecycleFallback && lifecycleStageFilterKeys.has(String(k))) return;
    appendFilter(k, v);
  });

  // Apply sort where supported (best-effort; backend will ignore if unsupported)
  try {
    if (sort && sort.key) {
      qs.set('order_by', String(sort.key));
      qs.set('order_dir', (String(sort.dir).toLowerCase() === 'desc') ? 'desc' : 'asc');
    }
  } catch {}

  // Choose path + id filter. Preserve the original endpoint map for all sections;
  // the lifecycle fallback only adds a timesheet-specific recovery path below.
  let path = '';
  if (section === 'timesheets') {
    path = '/api/timesheets/summary';
    qs.set('include_totals', 'false');
    qs.set('id', id); // backend supports OR(timesheet_id, contract_week_id)
    if (forceFresh) qs.set('_', String(Date.now()));
  } else if (section === 'invoices') {
    path = '/api/search/invoices';
    qs.set('ids', id);
  } else if (section === 'contracts') {
    path = '/api/contracts';
    qs.set('ids', id);
  } else if (section === 'candidates') {
    path = '/api/search/candidates';
    qs.set('ids', id);
  } else if (section === 'clients') {
    path = '/api/search/clients';
    qs.set('ids', id);
  } else if (section === 'umbrellas') {
    path = '/api/search/umbrellas';
    qs.set('ids', id);
  } else {
    path = `/api/search/${encodeURIComponent(section)}`;
    qs.set('ids', id);
  }

  const url = `${path}?${qs.toString()}`;

  try {
    const requestUrl = (typeof API === 'function') ? API(url) : url;
    const res = await authFetch(requestUrl, forceFresh ? { cache: 'no-store' } : undefined);
    if (!res || !res.ok) {
      const fallback = await fetchTimesheetLifecycleFallbackRow('summary-http-not-ok');
      window.__summaryCanonicalCache[cacheKey] = { t: now, row: fallback || null };
      return fallback || null;
    }

    const j = await res.json().catch(() => null);

    let rows = [];
    if (Array.isArray(j)) rows = j;
    else if (j && Array.isArray(j.items)) rows = j.items;
    else if (j && Array.isArray(j.rows))  rows = j.rows;
    else if (j && Array.isArray(j.data))  rows = j.data;

    const row = (rows && rows.length) ? (wantsTimesheetLifecycleFallback ? (rows.find((candidate) => sameCanonicalTimesheetId(candidate)) || rows[0]) : rows[0]) : null;
    if (!row || typeof row !== 'object' || Array.isArray(row)) {
      const fallback = await fetchTimesheetLifecycleFallbackRow('summary-returned-empty-items');
      window.__summaryCanonicalCache[cacheKey] = { t: now, row: fallback || null };
      return fallback || null;
    }

    // Normalizations to match renderSummary expectations
    try {
      if (section === 'timesheets') {
        if (!row.id) row.id = row.timesheet_id || row.current_timesheet_id || row.contract_week_id || id;
        if (!row.timesheet_id && row.current_timesheet_id) row.timesheet_id = row.current_timesheet_id;
        if (!row.current_timesheet_id && row.timesheet_id) row.current_timesheet_id = row.timesheet_id;
      }

      if (section === 'invoices') {
        if (!row.id) row.id = row.invoice_id || id;

        // Flatten joined client name if present
        if ((row.client_name == null || row.client_name === '') && row.client && typeof row.client === 'object') {
          row.client_name = row.client.name || '';
        }
      }
    } catch {}

    // Defensive exact-match validation: never return an unrelated row.
    try {
      let matchesRequestedId = false;

      if (section === 'timesheets') {
        matchesRequestedId = sameCanonicalTimesheetId(row);
      } else if (section === 'invoices') {
        const rowId = String(row.id || '').trim();
        const rowInvoiceId = String(row.invoice_id || '').trim();

        matchesRequestedId =
          (!!rowId && rowId === id) ||
          (!!rowInvoiceId && rowInvoiceId === id);
      } else {
        const rowId = String(row.id || '').trim();
        matchesRequestedId = !!rowId && rowId === id;
      }

      if (!matchesRequestedId) {
        const fallback = await fetchTimesheetLifecycleFallbackRow('summary-returned-wrong-row');
        window.__summaryCanonicalCache[cacheKey] = { t: now, row: fallback || null };
        return fallback || null;
      }
    } catch {
      const fallback = await fetchTimesheetLifecycleFallbackRow('summary-match-validation-failed');
      window.__summaryCanonicalCache[cacheKey] = { t: now, row: fallback || null };
      return fallback || null;
    }

    const finalRow = section === 'timesheets' && wantsTimesheetLifecycleFallback
      ? normaliseCanonicalTimesheetRow(row, 'summary-stage-safe')
      : row;
    window.__summaryCanonicalCache[cacheKey] = { t: now, row: finalRow };
    return finalRow;
  } catch (err) {
    try {
      if (wantsTimesheetLifecycleFallback) console.warn('[TS][SUMMARY_CANONICAL][FETCH_FAILED]', err?.message || err);
    } catch {}
    const fallback = await fetchTimesheetLifecycleFallbackRow('summary-fetch-threw');
    window.__summaryCanonicalCache[cacheKey] = { t: now, row: fallback || null };
    return fallback || null;
  }
}

async function refreshTimesheetsSummaryAfterRotation(newId, opts = {}) {
  const targetId = String(newId || '').trim();
  const blocking = !!(opts && opts.blocking === true);
  const afterLifecycleMutation = !!(opts && (opts.afterLifecycleMutation === true || opts.after_lifecycle_mutation === true));
  const background = !!(opts && (opts.background === true || opts.lazy === true || (afterLifecycleMutation && !blocking)));
  const lifecycleSequence = Number(opts?.lifecycleSequence ?? opts?.lifecycle_sequence ?? opts?.mutationSequence ?? opts?.mutation_sequence ?? 0) || 0;
  const sequenceAtEntry = (typeof readTimesheetLifecycleMutationSequence === 'function') ? readTimesheetLifecycleMutationSequence(window.modalCtx || {}) : Number(window.modalCtx?.__timesheetLifecycleMutationSeq || window.modalCtx?.__timesheetLatestLifecycleMutationSeq || 0) || 0;
  const isStaleLifecycleSummaryResult = () => {
    try {
      if (!afterLifecycleMutation || lifecycleSequence <= 0) return false;
      const live = (typeof readTimesheetLifecycleMutationSequence === 'function') ? readTimesheetLifecycleMutationSequence(window.modalCtx || {}) : Number(window.modalCtx?.__timesheetLifecycleMutationSeq || window.modalCtx?.__timesheetLatestLifecycleMutationSeq || 0) || 0;
      return live > lifecycleSequence;
    } catch { return false; }
  };
  const allowRenderAll = !!(opts && opts.allowRenderAll === true && !background);
  const skipTabRefresh = !!(opts && (opts.skipTabRefresh === true || background));
  const patchOnly = !!(opts && (opts.patchOnly === true || opts.patch_only === true || opts.bulkPatchOnly === true || opts.bulk_patch_only === true || opts.rowPatch || opts.row_patch || opts.data_row || opts.row));
  const preserveModalIdentity = !!(opts && (opts.preserveModalIdentity === true || opts.preserve_modal_identity === true || patchOnly));


  const buildOutcome = (ok, action, reason, extra = {}) => ({
    ok: !!ok,
    action: String(action || '').trim() || null,
    reason: String(reason || '').trim() || null,
    ...extra
  });



  if (background && afterLifecycleMutation && !blocking && opts.__background_worker !== true) {
    const scheduledOpts = {
      ...opts,
      background: false,
      lazy: false,
      __background_worker: true,
      skipTabRefresh: true,
      allowRenderAll: false,
      blocking: false,
      afterLifecycleMutation: true,
      lifecycleSequence: lifecycleSequence || sequenceAtEntry || null
    };
    setTimeout(() => {
      Promise.resolve(refreshTimesheetsSummaryAfterRotation(newId, scheduledOpts)).catch((err) => {
        try { console.warn('[TS][TRACE][SUMMARY_ROTATION] background refresh failed', err); } catch {}
      });
    }, Math.max(0, Math.min(Number(opts.debounce_ms ?? opts.debounceMs ?? 75) || 75, 500)));
    return buildOutcome(true, 'scheduled_background', 'after-lifecycle-background', {
      target_id: targetId || null,
      background: true,
      after_lifecycle_mutation: true,
      lifecycle_sequence: lifecycleSequence || sequenceAtEntry || null
    });
  }

  const normaliseApplyOutcome = (result, fallbackReason) => {
    if (result && typeof result === 'object') {
      const action = String(result.action || '').trim() || null;
      const ok = !!(
        result.ok === true ||
        result.removed === true ||
        ['patched', 'inserted', 'removed', 'renderAll', 'reloaded', 'targeted_refresh'].includes(action || '')
      );
      return {
        ok,
        action,
        reason: String(result.reason || fallbackReason || '').trim() || null,
        ...result
      };
    }

    if (result === true) {
      return buildOutcome(true, 'patched', fallbackReason || 'patched');
    }

    if (result === false) {
      return buildOutcome(false, null, fallbackReason || 'false');
    }

    if (typeof result === 'undefined') {
      return buildOutcome(false, null, fallbackReason || 'undefined');
    }

    return buildOutcome(!!result, null, fallbackReason || 'unknown-result');
  };

  const clone = (value) => {
    try { return JSON.parse(JSON.stringify(value)); } catch { return value; }
  };

  const makeRowKey = (rowLike) => {
    const row = (rowLike && typeof rowLike === 'object') ? rowLike : {};
    const explicit = String(row.row_key || row.new_row_key || '').trim();
    if (explicit) return explicit;
    const tsId = String(row.current_timesheet_id || row.timesheet_id || row.requested_timesheet_id || '').trim();
    if (tsId) return `timesheet:${tsId}`;
    const cwId = String(row.contract_week_id || row.contractWeekId || '').trim();
    if (cwId) return `contract_week:${cwId}`;
    const id = String(row.id || '').trim();
    if (id) return id;
    return '';
  };

  const mergePatchPayload = () => {
    const src = (opts && typeof opts === 'object') ? opts : {};
    const base = {};
    const dataRow = (src.data_row && typeof src.data_row === 'object') ? src.data_row : {};
    const rowObj = (src.row && typeof src.row === 'object') ? src.row : {};
    const rowPatch = (src.rowPatch && typeof src.rowPatch === 'object')
      ? src.rowPatch
      : ((src.row_patch && typeof src.row_patch === 'object') ? src.row_patch : {});
    const merged = { ...base, ...clone(rowObj), ...clone(dataRow), ...clone(rowPatch) };
    const resolvedTimesheetId = String(
      src.current_timesheet_id ||
      src.timesheet_id ||
      merged.current_timesheet_id ||
      merged.timesheet_id ||
      targetId ||
      ''
    ).trim();
    const resolvedContractWeekId = String(
      src.contract_week_id ||
      merged.contract_week_id ||
      ''
    ).trim();
    if (resolvedTimesheetId) {
      merged.timesheet_id = resolvedTimesheetId;
      merged.current_timesheet_id = resolvedTimesheetId;
      if (!merged.id || String(merged.id).trim().startsWith('contract_week:')) merged.id = resolvedTimesheetId;
    }
    if (resolvedContractWeekId) merged.contract_week_id = resolvedContractWeekId;
    const rowKey = String(src.new_row_key || src.row_key || merged.row_key || makeRowKey(merged) || '').trim();
    if (rowKey) merged.row_key = rowKey;
    const previousRowKey = String(src.previous_row_key || src.old_row_key || src.previousRowKey || merged.previous_row_key || '').trim();
    if (previousRowKey) merged.previous_row_key = previousRowKey;
    return merged;
  };

  const rowMatchesPatchIdentity = (row, patchRow, explicitRowKey = '') => {
    if (!row || typeof row !== 'object' || !patchRow || typeof patchRow !== 'object') return false;
    const candidateKey = String(row.row_key || makeRowKey(row) || '').trim();
    const targetKey = String(explicitRowKey || patchRow.row_key || makeRowKey(patchRow) || '').trim();
    if (candidateKey && targetKey && candidateKey === targetKey) return true;
    const rowId = String(row.id || '').trim();
    const rowTsId = String(row.timesheet_id || row.current_timesheet_id || '').trim();
    const rowCwId = String(row.contract_week_id || '').trim();
    const patchTsId = String(patchRow.timesheet_id || patchRow.current_timesheet_id || '').trim();
    const patchCwId = String(patchRow.contract_week_id || '').trim();
    if (patchTsId && (rowTsId === patchTsId || rowId === patchTsId)) return true;
    if (patchCwId && (rowCwId === patchCwId || rowId === patchCwId)) return true;
    return false;
  };

  const applyPatchToVisibleRows = (patchRow) => {
    const rows = Array.isArray(window.currentRows)
      ? window.currentRows
      : ((typeof currentRows !== 'undefined' && Array.isArray(currentRows)) ? currentRows : null);
    if (!rows || !patchRow || typeof patchRow !== 'object') return { changed: false, visible: false };
    const nextRowKey = String(patchRow.row_key || makeRowKey(patchRow) || '').trim();
    const previousRowKey = String(patchRow.previous_row_key || patchRow.old_row_key || '').trim();
    let changed = false;
    if (previousRowKey && previousRowKey !== nextRowKey) {
      for (let i = rows.length - 1; i >= 0; i -= 1) {
        const candidateKey = String(rows[i]?.row_key || makeRowKey(rows[i]) || '').trim();
        if (candidateKey && candidateKey === previousRowKey) {
          rows.splice(i, 1);
          changed = true;
        }
      }
    }
    const idx = rows.findIndex((row) => rowMatchesPatchIdentity(row, patchRow, nextRowKey));
    if (idx >= 0) {
      const normalised = (typeof normalizeSavedRecordForSummary === 'function')
        ? (normalizeSavedRecordForSummary('timesheets', patchRow) || patchRow)
        : patchRow;
      rows[idx] = { ...rows[idx], ...normalised };
      try { if (Array.isArray(window.currentRows)) window.currentRows = rows; } catch {}
      return { changed: true, visible: true };
    }
    try { if (Array.isArray(window.currentRows)) window.currentRows = rows; } catch {}
    return { changed, visible: false };
  };

  const applyPatchToModalIdentity = (patchRow) => {
    if (!preserveModalIdentity || !patchRow || typeof patchRow !== 'object') return false;
    const mc = (window.modalCtx && typeof window.modalCtx === 'object') ? window.modalCtx : null;
    if (!mc) return false;
    mc.data = (mc.data && typeof mc.data === 'object') ? mc.data : {};
    const tsId = String(patchRow.current_timesheet_id || patchRow.timesheet_id || targetId || '').trim();
    const cwId = String(patchRow.contract_week_id || '').trim();
    const rowKey = String(patchRow.row_key || makeRowKey(patchRow) || '').trim();
    if (tsId) {
      mc.data.timesheet_id = tsId;
      mc.data.id = tsId;
      mc.data.current_timesheet_id = tsId;
    }
    if (cwId) mc.data.contract_week_id = cwId;
    mc.formState = (mc.formState && typeof mc.formState === 'object') ? mc.formState : {};
    mc.formState.__forId = rowKey || tsId || cwId || targetId;
    mc.timesheetMeta = (mc.timesheetMeta && typeof mc.timesheetMeta === 'object') ? mc.timesheetMeta : {};
    if (tsId) {
      mc.timesheetMeta.expected_timesheet_id = tsId;
      mc.timesheetMeta.hasTs = true;
      mc.timesheetMeta.isPlannedWeek = false;
    }
    if (cwId) mc.timesheetMeta.contract_week_id = cwId;
    if (mc.timesheetDetails && typeof mc.timesheetDetails === 'object') {
      if (tsId) {
        mc.timesheetDetails.current_timesheet_id = tsId;
        mc.timesheetDetails.requested_timesheet_id = tsId;
        mc.timesheetDetails.expected_timesheet_id = tsId;
        mc.timesheetDetails.timesheet = (mc.timesheetDetails.timesheet && typeof mc.timesheetDetails.timesheet === 'object') ? mc.timesheetDetails.timesheet : {};
        mc.timesheetDetails.timesheet.timesheet_id = tsId;
      }
      if (cwId) mc.timesheetDetails.contract_week_id = cwId;
    }
    return true;
  };


  const patchModalLifecycleFromCanonicalRow = (rowPatch, options = {}) => {
    const patch = (rowPatch && typeof rowPatch === 'object') ? rowPatch : null;
    const mc = (window.modalCtx && typeof window.modalCtx === 'object') ? window.modalCtx : null;
    if (!patch || !mc || mc.entity !== 'timesheets') return false;
    if (isStaleLifecycleSummaryResult()) {
      try {
        console.log('[TS][TRACE][SUMMARY_ROTATION] skip-stale-modal-lifecycle-merge', {
          target_id: targetId,
          lifecycle_sequence: lifecycleSequence,
          live_sequence: (typeof readTimesheetLifecycleMutationSequence === 'function') ? readTimesheetLifecycleMutationSequence(mc) : Number(mc.__timesheetLifecycleMutationSeq || mc.__timesheetLatestLifecycleMutationSeq || 0) || 0
        });
      } catch {}
      return false;
    }

    const hasOwn = (obj, key) => !!(obj && typeof obj === 'object' && Object.prototype.hasOwnProperty.call(obj, key));
    const readFirst = (...values) => {
      for (const value of values) {
        if (value !== undefined && value !== null && String(value).trim() !== '') return value;
      }
      return null;
    };
    const readFirstOwn = (keys) => {
      for (const key of keys) {
        if (hasOwn(patch, key)) return patch[key];
      }
      return undefined;
    };
    const boolishNullable = (value) => {
      if (value === true || value === 1) return true;
      if (value === false || value === 0) return false;
      if (value == null || value === '') return null;
      const raw = String(value).trim().toLowerCase();
      if (['true', '1', 'yes', 'y', 'on'].includes(raw)) return true;
      if (['false', '0', 'no', 'n', 'off'].includes(raw)) return false;
      return null;
    };
    const copyOwn = (target, sourceKeys, targetKey) => {
      if (!target || typeof target !== 'object') return false;
      for (const key of sourceKeys) {
        if (hasOwn(patch, key)) {
          target[targetKey || key] = patch[key];
          return true;
        }
      }
      return false;
    };

    mc.data = (mc.data && typeof mc.data === 'object') ? mc.data : {};
    const patchTsId = String(patch.current_timesheet_id || patch.timesheet_id || patch.expected_timesheet_id || '').trim();
    const patchCwId = String(patch.contract_week_id || '').trim();
    const patchRowKey = String(patch.row_key || makeRowKey(patch) || '').trim();
    const modalMatches = !!(
      rowMatchesPatchIdentity(mc.data, patch, patchRowKey) ||
      (patchTsId && String(mc.timesheetDetails?.current_timesheet_id || mc.timesheetDetails?.timesheet_id || mc.timesheetDetails?.timesheet?.timesheet_id || '').trim() === patchTsId) ||
      (patchCwId && String(mc.timesheetDetails?.contract_week_id || mc.timesheetDetails?.contract_week?.id || '').trim() === patchCwId)
    );
    if (!modalMatches) return false;

    const explicitAuthorisedBool = (() => {
      const direct = boolishNullable(readFirstOwn(['is_authorised', 'authorised']));
      if (direct !== null) return direct;
      const actionFlags = (patch.action_flags && typeof patch.action_flags === 'object') ? patch.action_flags : null;
      return boolishNullable(actionFlags?.is_authorised ?? actionFlags?.authorised);
    })();
    const authServer = readFirstOwn(['authorised_at_server']);
    const authUtc = readFirstOwn(['authorised_at_utc']);
    const hasAuthServerOwn = hasOwn(patch, 'authorised_at_server');
    const hasAuthUtcOwn = hasOwn(patch, 'authorised_at_utc');
    const hasAuthTimestamp = !!(String(authServer == null ? '' : authServer).trim() || String(authUtc == null ? '' : authUtc).trim());
    const processingStatusU = String(patch.processing_status || patch.new_processing_status || '').trim().toUpperCase();
    const rowProvesUnauthorised = !!(
      patch.unauthorised === true ||
      patch.unauthorized === true ||
      explicitAuthorisedBool === false ||
      ((hasAuthServerOwn || hasAuthUtcOwn) && !hasAuthTimestamp && ['PENDING_AUTH', 'PENDING', 'UNAUTHORISED', 'UNAUTHORIZED'].includes(processingStatusU))
    );
    const rowProvesAuthorised = !!(explicitAuthorisedBool === true || hasAuthTimestamp);
    const resolvedTimesheetId = String(patchTsId || mc.data.current_timesheet_id || mc.data.timesheet_id || '').trim();
    const resolvedContractWeekId = String(patchCwId || mc.data.contract_week_id || mc.timesheetDetails?.contract_week_id || '').trim();
    const signature = String(readFirst(
      patch.backend_row_signature,
      patch.mutation_row_signature,
      patch.row_signature,
      patch.expected_row_signature
    ) || '').trim();

    const writeSurface = (surface, options = {}) => {
      if (!surface || typeof surface !== 'object') return false;
      if (resolvedTimesheetId) {
        surface.timesheet_id = resolvedTimesheetId;
        surface.current_timesheet_id = resolvedTimesheetId;
        if (options.setId === true) surface.id = resolvedTimesheetId;
      }
      if (resolvedContractWeekId) surface.contract_week_id = resolvedContractWeekId;

      if (rowProvesUnauthorised) {
        surface.authorised_at_server = null;
        surface.authorised_at_utc = null;
        surface.is_authorised = false;
        surface.authorised = false;
        if (hasOwn(surface, 'review_only')) surface.review_only = false;
      } else if (rowProvesAuthorised) {
        if (hasAuthServerOwn) surface.authorised_at_server = authServer || null;
        if (hasAuthUtcOwn) surface.authorised_at_utc = authUtc || null;
        surface.is_authorised = true;
        surface.authorised = true;
      } else {
        if (hasAuthServerOwn) surface.authorised_at_server = authServer || null;
        if (hasAuthUtcOwn) surface.authorised_at_utc = authUtc || null;
      }

      copyOwn(surface, ['processing_status', 'new_processing_status'], 'processing_status');
      copyOwn(surface, ['summary_stage'], 'summary_stage');
      copyOwn(surface, ['tools_stage'], 'tools_stage');
      copyOwn(surface, ['bulk_process_bucket'], 'bulk_process_bucket');
      copyOwn(surface, ['paid_at_utc'], 'paid_at_utc');
      copyOwn(surface, ['locked_by_invoice_id'], 'locked_by_invoice_id');
      copyOwn(surface, ['is_invoice_locked', 'invoice_locked'], 'is_invoice_locked');
      copyOwn(surface, ['is_paid', 'paid'], 'is_paid');
      copyOwn(surface, ['locked', 'is_locked'], 'is_locked');
      copyOwn(surface, ['review_only'], 'review_only');
      copyOwn(surface, ['expense_storage_target', 'expenseStorageTarget'], 'expense_storage_target');
      copyOwn(surface, ['expenseStorageTarget', 'expense_storage_target'], 'expenseStorageTarget');

      if (signature) {
        surface.backend_row_signature = signature;
        surface.mutation_row_signature = signature;
        surface.row_signature = signature;
        surface.expected_row_signature = signature;
      }
      return true;
    };

    writeSurface(mc.data, { setId: true });
    mc.timesheetDetails = (mc.timesheetDetails && typeof mc.timesheetDetails === 'object') ? mc.timesheetDetails : {};
    writeSurface(mc.timesheetDetails, { setId: false });

    mc.timesheetDetails.row = (mc.timesheetDetails.row && typeof mc.timesheetDetails.row === 'object') ? mc.timesheetDetails.row : {};

          target_id: targetId,
          modal_target_id: modalTargetId,
          current_tab_key: String(fr.currentTabKey || '').trim() || null,
          did_summary_patch: didSummaryPatch,
          did_targeted_refresh: didTargetedRefresh,
          did_render_all: didRenderAll,
          should_refresh_tab: shouldRefreshTab,
          skip_tab_refresh: skipTabRefresh,
          lifecycle_overview_tab_refresh: lifecycleOverviewTabNeedsRefresh
        });
      } catch {}

      if (shouldRefreshTab) {
        if (typeof fr.setTab === 'function' && fr.currentTabKey) {
          const prevSuppress = !!fr._suppressDirty;
          try {
            fr._suppressDirty = true;
            await fr.setTab(fr.currentTabKey);
            didTabRefresh = true;
          } finally {
            fr._suppressDirty = prevSuppress;
          }
        }
        fr._updateButtons && fr._updateButtons();
      } else {
        fr._updateButtons && fr._updateButtons();
      }
    }
  } catch {}

  if (finalOutcome && typeof finalOutcome === 'object') {
    finalOutcome.did_tab_refresh = !!didTabRefresh;
    finalOutcome.did_modal_merge = !!didModalMerge;
  }

  try {
    console.log('[TS][TRACE][SUMMARY_ROTATION] exit', {
      target_id: targetId,
      did_summary_patch: didSummaryPatch,
      did_targeted_refresh: didTargetedRefresh,
      did_render_all: didRenderAll,
      did_modal_merge: didModalMerge,
      did_tab_refresh: didTabRefresh,
      outcome: finalOutcome,
      modal_id_after: String(window.modalCtx?.data?.id || '').trim() || null,
      modal_timesheet_id_after: String(window.modalCtx?.data?.timesheet_id || '').trim() || null,
      modal_for_id_after: String(window.modalCtx?.formState?.__forId || '').trim() || null,
      expected_after: String(window.modalCtx?.timesheetMeta?.expected_timesheet_id || '').trim() || null
    });
  } catch {}

  return finalOutcome;
}
