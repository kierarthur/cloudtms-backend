import fs from'node:fs';import path from'node:path';import{compileManagedUnit}from'../../scripts/cloudtms-managed-release.mjs';import{compiledAtomicRehearsalUnit}from'../../scripts/healthroster-install1-full-rehearsal.mjs';import{mapExecutableSqlSource}from'../../scripts/cloudtms-db-release-lib.mjs';import{CAPABILITY_PREDECESSOR_MIGRATION as pin}from'../../scripts/healthroster-capability-predecessor-boundary.mjs';import{digest,literal,scanManagedSql}from'../../scripts/cloudtms-managed-release-sql.mjs';
export function capabilityCases(root,marker){
 const programSource=fs.readFileSync(path.join(root,pin.companionPath),'utf8');
 if(digest(programSource)!==pin.companionSha256)throw Error('CAPABILITY_PROGRAM_DRIFT');
 const parts=scanManagedSql(programSource).statements;
 if(parts.length!==2)throw Error('PROGRAM_SHAPE');
 const program=programSource.slice(parts[1].start,parts[1].end),lock=programSource.slice(parts[0].start,parts[0].end);
 const unit=compiledAtomicRehearsalUnit(compileManagedUnit({path:pin.path,sha256:pin.contentSha256},'migration',{root,mapSource:mapExecutableSqlSource}));
 if(!unit.atomicCapabilityPredecessorCompanion||!unit.sql.includes(program))throw Error('COMPILED_PREBODY_ABSENT');
 const table='public.weekly_source_group_rule_assignments',oldNames=['weekly_source_group_rule_assignments_feature_purpose_check','weekly_source_group_rule_assignments_check2'],newNames=['weekly_source_assignment_feature_purpose_v2','weekly_source_assignment_feature_version_v2'];
 const matrix=JSON.parse(fs.readFileSync(path.join(root,pin.expressionPath),'utf8')).checkTruthMatrix;
 if(matrix.length!==10)throw Error('TRUTH_MATRIX_CHANGED');
 const snapshot="select coalesce(jsonb_agg(jsonb_build_object('name',c.conname,'oid',c.oid::bigint,'definition',pg_catalog.pg_get_constraintdef(c.oid),'expression',pg_catalog.pg_get_expr(c.conbin,c.conrelid),'key',c.conkey,'validated',c.convalidated,'local',c.conislocal,'noinherit',c.connoinherit,'deferrable',c.condeferrable,'deferred',c.condeferred,'inhcount',c.coninhcount,'parent',c.conparentid::bigint)order by c.conname),'[]'::jsonb) from pg_catalog.pg_constraint c where c.conrelid="+literal(table)+"::regclass and c.contype='c'";
 const capture="insert into private."+marker+"(marker,checks_before)values(true,("+snapshot+"));";
 function truth(phase,names){let b="do $hr_truth_"+phase+"$ declare purpose_expr text; version_expr text; purpose_value boolean; version_value boolean;begin ";
 b+="select pg_catalog.pg_get_expr(c.conbin,c.conrelid)into strict purpose_expr from pg_catalog.pg_constraint c where c.conrelid="+literal(table)+"::regclass and c.conname="+literal(names[0])+";";
 b+="select pg_catalog.pg_get_expr(c.conbin,c.conrelid)into strict version_expr from pg_catalog.pg_constraint c where c.conrelid="+literal(table)+"::regclass and c.conname="+literal(names[1])+";";
 for(const [i,row]of matrix.entries()){const expected=row[phase+'Checks'];const v=row.rule_set_version_id===null?'null::uuid':"'11111111-1111-4111-8111-111111111111'::uuid";const from=" from(values("+(row.feature_purpose===null?'null':literal(row.feature_purpose))+"::text,"+v+"))as x(feature_purpose,rule_set_version_id)";
 b+="execute 'select ('||purpose_expr||')' ||"+literal(from)+" into purpose_value;execute 'select ('||version_expr||')'||"+literal(from)+" into version_value;";
 b+="if purpose_value is distinct from "+(expected.PURPOSE===null?'null':literal(expected.PURPOSE))+"::boolean or version_value is distinct from "+(expected.VERSION===null?'null':literal(expected.VERSION))+"::boolean then raise exception "+literal('CAPABILITY_TRUTH_'+phase+'_'+i)+" using errcode='XX000';end if;";
 }return b+"end;$hr_truth_"+phase+"$;";}
 const cases=[];let negative="do $hr_cap_negative$ declare got boolean; definition text;begin ";
 for(const name of oldNames)for(const kind of['RENAMED_CHECK','UNVALIDATED_CHECK','EXTRA_CHECK']){
 const id=name+'#'+kind;let mutate='';
 if(kind==='RENAMED_CHECK')mutate="execute "+literal('alter table '+table+' rename constraint '+name+' to hr_capability_negative')+";";
 if(kind==='UNVALIDATED_CHECK')mutate="select pg_catalog.pg_get_constraintdef(c.oid)into strict definition from pg_catalog.pg_constraint c where c.conrelid="+literal(table)+"::regclass and c.conname="+literal(name)+";execute "+literal('alter table '+table+' drop constraint '+name)+";execute "+literal('alter table '+table+' add constraint '+name+' ')+"||definition||' not valid';";
 if(kind==='EXTRA_CHECK')mutate="execute "+literal("alter table "+table+" add constraint hr_capability_negative check(feature_purpose in('RULE_SET','SIGNED_CHECK_ONLY')) not valid")+";";
 negative+="got:=false;begin "+mutate+"begin execute "+literal(program)+";exception when sqlstate '55000' then if sqlerrm<>'HEALTHROSTER_CAPABILITY_PREDECESSOR_CHECKS_CHANGED' then raise;end if;got:=true;end;if not got then raise exception "+literal('CAPABILITY_DRIFT_ACCEPTED:'+id)+" using errcode='XX000';end if;raise exception 'HR_CAPABILITY_NEGATIVE_ROLLBACK' using errcode='Z0001';exception when sqlstate 'Z0001' then if sqlerrm<>'HR_CAPABILITY_NEGATIVE_ROLLBACK' then raise;end if;end;if not got then raise exception 'CAPABILITY_REFUSAL_RESULT_LOST';end if;";cases.push({id,expectedError:'HEALTHROSTER_CAPABILITY_PREDECESSOR_CHECKS_CHANGED'});
 }negative+="end;$hr_cap_negative$;";
 const storeAfter="update private."+marker+" set checks_after=("+snapshot+");";
 const oldArray=oldNames.map(literal).join(','),newArray=newNames.map(literal).join(',');
 const assertAfter="do $hr_cap_after$ declare b jsonb;a jsonb;r_old jsonb;r_new jsonb;got boolean:=false;begin select checks_before,checks_after into strict b,a from private."+marker+";if jsonb_array_length(b)<>10 or jsonb_array_length(a)<>10 then raise exception 'CAPABILITY_TOTAL_CHECKS_CHANGED';end if;select jsonb_agg(x order by x->>'name')into r_old from jsonb_array_elements(b)x where x->>'name' not in("+oldArray+");select jsonb_agg(x order by x->>'name')into r_new from jsonb_array_elements(a)x where x->>'name' not in("+newArray+");if jsonb_array_length(r_old)<>8 or r_old is distinct from r_new then raise exception 'CAPABILITY_RETAINED_CHECKS_CHANGED';end if;if exists(select 1 from jsonb_array_elements(a)x where x->>'name' in("+oldArray+"))or(select count(*)from jsonb_array_elements(a)x where x->>'name' in("+newArray+") and (x->>'validated')::boolean and (x->>'local')::boolean and not(x->>'noinherit')::boolean and not(x->>'deferrable')::boolean and not(x->>'deferred')::boolean and (x->>'inhcount')::int=0 and (x->>'parent')::bigint=0)<>2 then raise exception 'CAPABILITY_SUCCESSOR_CHECK_SET_CHANGED';end if;begin execute "+literal(program)+";exception when sqlstate '55000' then if sqlerrm<>'HEALTHROSTER_CAPABILITY_PREDECESSOR_CHECKS_CHANGED' then raise;end if;got:=true;end;if not got then raise exception 'CAPABILITY_ALREADY_EXTENDED_NOT_REFUSED';end if;end;$hr_cap_after$;";
 return{sql:[lock,program,capture,truth('old',oldNames),negative,unit.sql,storeAfter,assertAfter,truth('new',newNames),"select jsonb_build_object('status','PASS_CAPABILITY_PREDECESSOR_BODY_TRUTHS_AND_REFUSALS','driftCases',6,'alreadyExtendedRefusals',1,'truthClassesBefore',10,'truthClassesAfter',10,'retainedChecksUnchanged',8,'nativeChecksBefore',checks_before,'nativeChecksAfter',checks_after) from private."+marker+";"].join('\n'),cases,truthClasses:matrix,sourceProgramSha256:digest(programSource),compiledCapabilityUnit:(({sql,...rest})=>rest)(unit)};
}
