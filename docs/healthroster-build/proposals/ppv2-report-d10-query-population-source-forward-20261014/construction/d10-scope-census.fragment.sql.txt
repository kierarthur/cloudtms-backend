  -- D10 adds only actual Admin-visible native query scopes. It does not make a
  -- missing Source report, a completed final revision or an import-ready claim.
  select lower(btrim(coalesce(actor.role,'')))='admin' into v_admin
  from public.tms_users actor where actor.id=v_actor and actor.is_active;
  if v_admin then
    for v_d10_scope in
      select distinct g.id group_id,g.display_name source_name,gc.client_id,cl.name client_name,
        c.id cycle_id,c.scope_client_id,c.finalisation_week_ending,c.cutoff_at_utc,c.version cycle_version
      from public.weekly_source_groups g
      join public.weekly_source_cycles c on c.source_group_id=g.id
      join public.weekly_source_group_clients gc on gc.source_group_id=g.id
      join public.clients cl on cl.id=gc.client_id
      where g.active and g.source_family='ROSTER'
        and (v_group is null or g.id=v_group) and (v_client is null or gc.client_id=v_client)
        and (v_week is null or c.finalisation_week_ending=v_week)
        and (c.scope_client_id is null or c.scope_client_id=gc.client_id)
        and (c.scope_client_id is not null or not exists(select 1 from public.weekly_source_cycles preferred
          where preferred.source_group_id=g.id and preferred.scope_client_id=gc.client_id
            and preferred.finalisation_week_ending=c.finalisation_week_ending))
        and exists(select 1 from public.timesheets t
          join public.weekly_source_client_policies p on p.source_group_id=g.id and p.client_id=gc.client_id
          where t.week_ending_date between c.finalisation_week_ending-6 and c.finalisation_week_ending
            and t.week_ending_date between gc.valid_from and coalesce(gc.valid_to,'infinity'::date)
            and t.week_ending_date between p.effective_from and coalesce(p.effective_to,'infinity'::date)
            and p.authority_mode='TIMESHEET_AUTHORITY' and p.document_mode='INVOICE_EVIDENCE_REQUIRED' and not p.self_bill_enabled)
        -- Source candidates above keep their existing row and report totals.
        and not exists(select 1 from public.weekly_source_client_policies p where p.source_group_id=g.id and p.client_id=gc.client_id
          and p.authority_mode='SOURCE_AUTHORITY' and c.finalisation_week_ending between p.effective_from and coalesce(p.effective_to,'infinity'::date))
      order by g.id,gc.client_id,c.finalisation_week_ending,c.id
    loop
      -- Enumerate real native weeks inside this exact cycle. Eligibility and
      -- parent identity still come exclusively from the same shared D10 owner.
      with native_scope_frames as materialized (
        select native_week.week_ending_date,private.weekly_source_check_only_query_groups_v1(v_d10_scope.cycle_id,v_d10_scope.client_id,
          jsonb_build_object('actor_user_id',v_actor,'source_group_id',v_d10_scope.group_id,'week_ending',native_week.week_ending_date)) value
        from (select distinct t.week_ending_date from public.timesheets t
          where t.week_ending_date between v_d10_scope.finalisation_week_ending-6 and v_d10_scope.finalisation_week_ending
            and exists(select 1 from public.weekly_source_group_clients gc where gc.source_group_id=v_d10_scope.group_id
              and gc.client_id=v_d10_scope.client_id and t.week_ending_date between gc.valid_from and coalesce(gc.valid_to,'infinity'::date))) native_week
      )
      select jsonb_build_object('revisions',coalesce(jsonb_agg(jsonb_build_object(
        'week_ending',native_frame.week_ending_date,'read_revision',native_frame.value->>'read_revision',
        'source_read_revision',native_frame.value->>'source_read_revision') order by native_frame.week_ending_date),'[]'::jsonb),
        'group_count',coalesce(sum(jsonb_array_length(native_frame.value->'groups')),0),
        'valid',coalesce(bool_and(coalesce(native_frame.value->>'contract'='WEEKLY_SOURCE_CHECK_ONLY_QUERY_GROUPS_V1'
          and native_frame.value->>'actor_user_id'=v_actor::text and native_frame.value->>'permission_view'='ADMIN'
          and native_frame.value#>>'{scope,source_group_id}'=v_d10_scope.group_id::text
          and native_frame.value#>>'{scope,source_cycle_id}'=v_d10_scope.cycle_id::text
          and native_frame.value#>>'{scope,client_id}'=v_d10_scope.client_id::text
          and native_frame.value#>>'{scope,week_ending}'=native_frame.week_ending_date::text
          and jsonb_typeof(native_frame.value->'groups')='array'
          and native_frame.value->'source_outreach_eligible'='false'::jsonb and native_frame.value->'bulk_actions'='{}'::jsonb
          and coalesce(native_frame.value->>'read_revision','')~'^[0-9a-f]{64}$'
          and coalesce(native_frame.value->>'source_read_revision','')~'^[0-9a-f]{64}$',false)),false))
      into v_d10_frame
      from native_scope_frames native_frame;
      if v_d10_frame->'valid' is distinct from 'true'::jsonb then
        raise exception 'WEEKLY_SOURCE_CHECK_ONLY_NATIVE_GROUP_READ_UNAVAILABLE' using errcode='55000';end if;
      if (v_d10_frame->>'group_count')::integer>0 then
        v_d10_scopes:=v_d10_scopes||jsonb_build_array(jsonb_build_object(
          'group_id',v_d10_scope.group_id,'source_name',v_d10_scope.source_name,'client_id',v_d10_scope.client_id,'client_name',v_d10_scope.client_name,
          'cycle_id',v_d10_scope.cycle_id,'scope_client_id',v_d10_scope.scope_client_id,'finalisation_week_ending',v_d10_scope.finalisation_week_ending,
          'cutoff_at_utc',v_d10_scope.cutoff_at_utc,'cycle_version',v_d10_scope.cycle_version,
          'query_read_revision',encode(private.weekly_source_sha256_jsonb_v1('WEEKLY_SOURCE_D10_SCOPE_CENSUS_V1',
            jsonb_build_object('actor_user_id',v_actor,'source_cycle_id',v_d10_scope.cycle_id,'client_id',v_d10_scope.client_id,
              'native_read_revisions',v_d10_frame->'revisions')),'hex')));
      end if;
    end loop;
  end if;
