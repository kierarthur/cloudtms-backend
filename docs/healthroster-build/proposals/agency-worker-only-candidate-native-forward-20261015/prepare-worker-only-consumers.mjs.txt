function owner(file, before, changes) {
  const nl = before.includes('\r\n') ? '\r\n' : '\n';
  const lines = value => value.replaceAll('\r\n', '\n').replaceAll('\n', nl);
  let after = before;
  const hunks = changes.map(change => ({ ...change, before: lines(change.before), after: lines(change.after) }));
  for (const hunk of hunks) {
    assert.equal(after.split(hunk.before).length, 2, file + ': ' + hunk.purpose);
    after = after.replace(hunk.before, () => hunk.after);
  }
  fs.writeFileSync(path.join(here, 'before', file), before);
  fs.writeFileSync(path.join(here, 'forward', file), after);
  owners.push({ file, before_sha256: sha(before), after_sha256: sha(after), hunks });
  return after;
}
