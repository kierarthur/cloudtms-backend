-- Queue health snapshot.
SELECT
  status,
  job_type,
  COUNT(*) AS job_count,
  MIN(created_at_utc) AS oldest_created_at_utc,
  MIN(run_at_utc) AS oldest_run_at_utc,
  MAX(updated_at_utc) AS newest_updated_at_utc
FROM public.banking_pay_workbench_jobs
WHERE created_at_utc >= now() - interval '2 hours'
GROUP BY status, job_type
ORDER BY status, job_type;

-- Failed/dead job details.
SELECT
  id, session_id, candidate_id, job_type, status, attempt_count, max_attempts,
  run_at_utc, started_at_utc, completed_at_utc, failed_at_utc, created_at_utc, updated_at_utc,
  last_error_json, payload_json
FROM public.banking_pay_workbench_jobs
WHERE created_at_utc >= now() - interval '2 hours'
  AND UPPER(BTRIM(COALESCE(status, ''))) IN ('FAILED', 'DEAD')
ORDER BY failed_at_utc DESC NULLS LAST, updated_at_utc DESC
LIMIT 50;

-- Stale running jobs.
SELECT
  id, session_id, candidate_id, job_type, status, started_at_utc, updated_at_utc,
  payload_json->>'worker_id' AS worker_id,
  payload_json->>'worker_lease_expires_at_utc' AS worker_lease_expires_at_utc,
  payload_json
FROM public.banking_pay_workbench_jobs
WHERE UPPER(BTRIM(COALESCE(status, ''))) = 'RUNNING'
  AND updated_at_utc < now() - interval '2 minutes'
ORDER BY updated_at_utc ASC
LIMIT 50;

-- Session progress health for recent sessions.
SELECT
  s.id, s.status, s.version, s.pay_date, s.created_at_utc, s.updated_at_utc,
  s.progress_state, s.progress_json->'blocker_codes' AS blocker_codes,
  s.progress_json->'blocker_counts' AS blocker_counts,
  s.preview_row_count, s.selected_row_count, s.scope_total_count, s.scope_ready_count,
  public.pay_workbench_session_get_progress_light(s.id) AS live_progress
FROM public.banking_pay_workbench_sessions s
WHERE s.created_at_utc >= now() - interval '2 hours'
   OR s.updated_at_utc >= now() - interval '2 hours'
ORDER BY s.updated_at_utc DESC
LIMIT 20;

-- Dirty/current source-row mismatch check.
SELECT
  session_id, session_version, candidate_id, timesheet_id, line_key,
  COUNT(*) FILTER (WHERE status = 'CURRENT') AS current_count,
  COUNT(*) FILTER (WHERE status = 'DIRTY') AS dirty_count,
  COUNT(*) FILTER (WHERE status = 'SUPERSEDED') AS superseded_count,
  MIN(source_change_seq) AS min_source_change_seq,
  MAX(source_change_seq) AS max_source_change_seq,
  ARRAY_AGG(DISTINCT source_build_run_id) FILTER (WHERE source_build_run_id IS NOT NULL) AS source_build_run_ids
FROM public.banking_pay_workbench_candidate_source_lines
WHERE updated_at_utc >= now() - interval '2 hours'
GROUP BY session_id, session_version, candidate_id, timesheet_id, line_key
HAVING COUNT(*) FILTER (WHERE status = 'CURRENT') > 0
   AND COUNT(*) FILTER (WHERE status = 'DIRTY') > 0
ORDER BY session_id, candidate_id, timesheet_id, line_key;

-- Progress blocker aggregate across recent sessions.
WITH recent_sessions AS (
  SELECT id
  FROM public.banking_pay_workbench_sessions
  WHERE created_at_utc >= now() - interval '2 hours'
     OR updated_at_utc >= now() - interval '2 hours'
), progress AS (
  SELECT id AS session_id, public.pay_workbench_session_get_progress_light(id) AS p
  FROM recent_sessions
)
SELECT
  p->>'phase' AS phase,
  p->'job_counts'->>'unresolved_failed' AS unresolved_failed,
  p->'job_counts'->>'unresolved_dead' AS unresolved_dead,
  p->'blocker_counts'->>'source_rows_dirty' AS source_rows_dirty,
  p->'blocker_counts'->>'queued_jobs' AS queued_jobs,
  p->'blocker_counts'->>'running_jobs' AS running_jobs,
  COUNT(*) AS session_count
FROM progress
GROUP BY
  p->>'phase',
  p->'job_counts'->>'unresolved_failed',
  p->'job_counts'->>'unresolved_dead',
  p->'blocker_counts'->>'source_rows_dirty',
  p->'blocker_counts'->>'queued_jobs',
  p->'blocker_counts'->>'running_jobs'
ORDER BY session_count DESC;
